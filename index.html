import { SeatmapBookingRenderer }
  from '@seatmap.pro/renderer-ce';
import schema from './schema.json';
import prices from './prices.json';

const renderer = new SeatmapBookingRenderer(
  element, {
    onSeatSelect: seat => reserveSeat(seat),
    onCartChange: cart => updateCart(cart),
  }
);

await renderer.loadData(schema, prices);
